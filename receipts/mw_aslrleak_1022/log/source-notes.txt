		char *p;

			}

			p = ZSTR_VAL(new_value);
		}

		if (PG(open_basedir) && *p && php_check_open_basedir(p)) {
			return FAILURE;
		}
	}

=== note ===
MediaWiki installs PHPSessionHandler and disables PHP session cookies (session.use_cookies=0).
PHP's upload-progress handler runs before that install and reads PHPSESSID into a files-handler
session buffer from session.save_path; that is the path used by the prior zif_passthru record.
The MediaWiki login/token cookie (wikidb_session) is a different, MediaWiki-managed session ID.
